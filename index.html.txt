    <script>
        // Array para almacenar las inscripciones temporalmente
        let inscripciones = [];

        // Lógica del Formulario
        document.getElementById('registroForm').addEventListener('submit', function(e) {
            e.preventDefault();
            
            const nuevaInscripcion = {
                id: Date.now(), // Identificador único
                nombre: document.getElementById('nombre').value,
                rut: document.getElementById('rut').value,
                telefono: document.getElementById('telefono').value,
                correo: document.getElementById('correo').value,
                club: document.getElementById('club').value,
                caballo: document.getElementById('caballo').value,
                serie: document.getElementById('serie').value,
                companero: document.getElementById('companero').value
            };

            inscripciones.push(nuevaInscripcion);
            actualizarTabla();
            this.reset(); // Limpia el formulario
            alert("¡Inscripción guardada correctamente!");
        });

        // Lógica del Panel de Administración (Tabla y Conteos)
        function actualizarTabla() {
            const tbody = document.querySelector('#tablaInscripciones tbody');
            tbody.innerHTML = '';
            let conteo = {};

            inscripciones.forEach(ins => {
                // Registrar conteos por serie
                conteo[ins.serie] = (conteo[ins.serie] || 0) + 1;

                // Crear fila en la tabla
                const tr = document.createElement('tr');
                tr.innerHTML = `
                    <td>${ins.serie}</td>
                    <td>${ins.nombre}</td>
                    <td>${ins.companero}</td>
                    <td>${ins.caballo}</td>
                    <td>${ins.club}</td>
                    <td>${ins.telefono}</td>
                    <td>
                        <button onclick="eliminarInscripcion(${ins.id})" class="delete-btn">Eliminar</button>
                    </td>
                `;
                tbody.appendChild(tr);
            });

            // Actualizar panel de conteo por serie
            let conteoHtml = "<strong>Inscritos por serie:</strong><br>";
            if(Object.keys(conteo).length === 0) {
                conteoHtml += "No hay inscripciones aún.";
            } else {
                for (let s in conteo) {
                    conteoHtml += `• ${s}: ${conteo[s]} inscrito(s)<br>`;
                }
            }
            document.getElementById('conteoSeries').innerHTML = conteoHtml;
        }

        // Lógica para Eliminar
        function eliminarInscripcion(id) {
            if(confirm("¿Estás seguro de eliminar esta inscripción?")) {
                inscripciones = inscripciones.filter(ins => ins.id !== id);
                actualizarTabla();
            }
        }

        // Lógica para Buscar/Filtrar
        function filtrarTabla() {
            const filtro = document.getElementById('buscar').value.toLowerCase();
            const filas = document.querySelectorAll('#tablaInscripciones tbody tr');
            filas.forEach(fila => {
                // Filtramos por todo el texto de la fila (nombre, serie, etc.)
                const texto = fila.innerText.toLowerCase();
                fila.style.display = texto.includes(filtro) ? '' : 'none';
            });
        }

        // Lógica para Exportar a Excel
        function exportarExcel() {
            if(inscripciones.length === 0) {
                alert("No hay datos para exportar.");
                return;
            }

            // Mapeamos los datos para tener solo las columnas solicitadas
            const datosExportar = inscripciones.map(ins => ({
                Serie: ins.serie,
                Jinete: ins.nombre,
                Compañero: ins.companero,
                Caballo: ins.caballo,
                Club: ins.club,
                Teléfono: ins.telefono
            }));
            
            // Usamos SheetJS para construir el archivo
            const ws = XLSX.utils.json_to_sheet(datosExportar);
            const wb = XLSX.utils.book_new();
            XLSX.utils.book_append_sheet(wb, ws, "Inscripciones");
            
            // Descargar archivo
            XLSX.writeFile(wb, "Inscripciones_Rodeo.xlsx");
        }
